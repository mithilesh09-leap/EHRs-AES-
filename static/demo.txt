from flask import Flask, render_template_string, request
import sqlite3
import os
import hashlib
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend
from base64 import b64encode, b64decode

app = Flask(__name__)

# AES256 encryption key
SECRET_KEY = hashlib.sha256("your-secret-key-here".encode()).digest()

# Create a connection to SQLite database
def create_connection():
    conn = sqlite3.connect(':memory:')  # In-memory SQLite database (no need for file storage)
    return conn

# Encrypt data using AES256 encryption
def encrypt_data(plain_text, key):
    iv = os.urandom(16)  # 16-byte initialization vector
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    encryptor = cipher.encryptor()

    # Padding plaintext to be a multiple of block size (16 bytes)
    padding_length = 16 - len(plain_text) % 16
    padded_plain_text = plain_text + chr(padding_length) * padding_length
    encrypted_data = encryptor.update(padded_plain_text.encode()) + encryptor.finalize()

    return b64encode(iv + encrypted_data).decode('utf-8')

# Decrypt data using AES256 encryption
def decrypt_data(encrypted_text, key):
    encrypted_data = b64decode(encrypted_text)
    iv, encrypted_data = encrypted_data[:16], encrypted_data[16:]

    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    decryptor = cipher.decryptor()

    decrypted_data = decryptor.update(encrypted_data) + decryptor.finalize()

    # Remove padding
    padding_length = decrypted_data[-1]
    return decrypted_data[:-padding_length].decode('utf-8')

# Create database table
def create_table():
    conn = create_connection()
    cursor = conn.cursor()
    cursor.execute("""
    CREATE TABLE IF NOT EXISTS patients (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        age INTEGER NOT NULL,
        medical_history TEXT
    );
    """)
    conn.commit()
    conn.close()

# Insert a new patient record into the database
def insert_patient(name, age, medical_history, key):
    encrypted_medical_history = encrypt_data(medical_history, key)
    conn = create_connection()
    cursor = conn.cursor()
    cursor.execute("""
    INSERT INTO patients (name, age, medical_history)
    VALUES (?, ?, ?)
    """, (name, age, encrypted_medical_history))
    conn.commit()
    conn.close()

# Get patient record from the database by ID
def get_patient(id, key):
    conn = create_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM patients WHERE id=?", (id,))
    patient = cursor.fetchone()
    conn.close()
    if patient:
        decrypted_medical_history = decrypt_data(patient[3], key)
        return {
            'id': patient[0],
            'name': patient[1],
            'age': patient[2],
            'medical_history': decrypted_medical_history
        }
    return None
